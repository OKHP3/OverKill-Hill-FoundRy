- Backup roundtrip and generated package comparison: passed with the command below. It parses the backup using the real parser, checks the active project, regenerates files, and compares every generated file byte-for-byte.

```bash
node --experimental-strip-types - <<'NODE'
import { readFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { buildCapabilityFiles, parseCapabilityBackup } from './artifacts/mockup-sandbox/src/lib/capability-workbench.ts';
const root = 'examples/workbench/skill-example';
const workspace = parseCapabilityBackup(readFileSync(`${root}/backup.json`, 'utf8'));
if (workspace.projects.length !== 1 || workspace.projects[0].kind !== 'skill') throw new Error('unexpected backup');
for (const file of buildCapabilityFiles(workspace.projects[0])) {
  const actual = readFileSync(`${root}/generated/${file.path}`, 'utf8');
  if (actual !== file.content) throw new Error(`generated mismatch: ${file.path}`);
}
console.log('F02 roundtrip OK', createHash('sha256').update(readFileSync(`${root}/backup.json`)).digest('hex'));
NODE
```
