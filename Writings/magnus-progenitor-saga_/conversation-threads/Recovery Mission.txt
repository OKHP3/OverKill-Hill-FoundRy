            const cells = Array.from(row.children).map(td => td.innerText.trim());
            out += '| ' + cells.join(' | ') + ' |\n';
            if (i === 0) out += '| ' + cells.map(() => '---').join(' | ') + ' |\n';
          });
          out += '\n';
        } else {
          out += el.innerText + '\n\n';
        }
      });
      return out.trim();
    };

    }

    await delay(20); // allow DOM to settle
  }
