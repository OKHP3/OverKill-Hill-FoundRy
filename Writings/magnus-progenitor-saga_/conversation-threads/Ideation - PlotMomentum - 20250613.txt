process_data = []
for name, content in process_blocks:
    process_name = name.strip()
    procedures = re.findall(r'Procedure[s]?:\s*(.*?)\n', content)
    if not procedures:
        # Check for implied steps in content body if no formal "Procedures:" label found
        steps = re.findall(r'^\s*\d+\.\s+(.*)', content, re.MULTILINE)
        for step in steps:
            if '(' in step and ')' in step:  # Rough match for procedural call
                process_data.append({"Process": process_name, "Procedure": step.strip()})

                process_data.append({"Process": process_name, "Procedure": "Implied: " + step.strip()})
    else:
        for proc in procedures:
            for p in proc.split(','):
                process_data.append({"Process": process_name, "Procedure": p.strip()})
